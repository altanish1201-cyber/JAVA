10. Write a Java program to demonstrate the use of abstract classes and abstract methods.
abstract class Shape {
    abstract void draw();
    void message() {
        System.out.println("Drawing Shape");
    }
}

class Circle extends Shape {
    void draw() {
        System.out.println("Drawing Circle");
    }
}

public class AbstractDemo {
    public static void main(String[] args) {
        Circle c = new Circle();
        c.message();
        c.draw();
    }
}

11. Write a Java program to demonstrate the implementation of interfaces.
interface Animal {
    void eat();
    void sleep();
}

}

    }
}

8. Write a Java program to demonstrate the use of the super keyword in inheritance.
class Animal {
    String name;
    Animal(String name) {
        this.name=name;
    }
    void sound() {
        System.out.println("Animal makes sound");
    }
}

class Dog extends Animal {
    String name;
    Dog(String name) {
        this.name=name;
        super("bird");
    }
    void display() {
        System.out.println("Parent Name: " + super.name);
        System.out.println("Child Name: " + name);
        super.sound();
    }
}

public class SuperDemo {
    public static void main(String[] args) {
        Dog d = new Dog("Dog");
        d.display();
    }
}
9. Write a Java program to demonstrate the use of the final keyword with variables, methods, and classes.
final class College {

6. Write a Java program to demonstrate the use of an inner class and an anonymous class.
class Outer {
    private String message = "Welcome to Java Programming";
    // Inner Class
    class Inner {
        void display() {
            System.out.println("Message: " + message);
        }
    }
}

interface Greeting {
    void sayHello();
}

public class InnerAnonymousDemo {
    public static void main(String[] args) {
        // Inner Class
        Outer outer = new Outer();
        Outer.Inner inner = outer.new Inner();
        inner.display();

        // Anonymous Class
        Greeting greet = new Greeting() {
            public void sayHello() {
                System.out.println("Hello from Anonymous Class");
            }
        };
        greet.sayHello();
    }
}
